cmake_minimum_required(VERSION 3.24)
project(VANTAGE VERSION 0.2.0 LANGUAGES CXX)
option(VANTAGE_CUDA "Build CUDA backend" OFF)
option(VANTAGE_OPENMP "Use OpenMP" ON)
option(VANTAGE_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(vantage src/model.cpp src/io.cpp src/verify.cpp src/preprocess.cpp src/cpu.cpp src/solver.cpp src/mip.cpp)
target_include_directories(vantage PUBLIC include PRIVATE third_party)
target_compile_options(vantage PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>)
if(VANTAGE_OPENMP)
 find_package(OpenMP)
 if(OpenMP_CXX_FOUND)
  target_link_libraries(vantage PUBLIC OpenMP::OpenMP_CXX)
 endif()
endif()
if(VANTAGE_CUDA)
 enable_language(CUDA)
 find_package(CUDAToolkit REQUIRED)
 target_sources(vantage PRIVATE src/cuda.cu)
 target_compile_definitions(vantage PUBLIC VANTAGE_HAS_CUDA)
 target_link_libraries(vantage PRIVATE CUDA::cudart CUDA::cusparse)
 set_target_properties(vantage PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
else()
 target_sources(vantage PRIVATE src/cuda_stub.cpp)
endif()
if(VANTAGE_SANITIZE)
 target_compile_options(vantage PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(vantage PUBLIC -fsanitize=address,undefined)
endif()
add_executable(vantage_cli app/main.cpp)
set_target_properties(vantage_cli PROPERTIES OUTPUT_NAME vantage)
target_include_directories(vantage_cli PRIVATE third_party)
target_link_libraries(vantage_cli PRIVATE vantage)
enable_testing()
add_executable(vantage_tests tests/test_core.cpp)
target_link_libraries(vantage_tests PRIVATE vantage)
add_test(NAME core COMMAND vantage_tests)
set_tests_properties(core PROPERTIES WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
install(TARGETS vantage vantage_cli)
install(DIRECTORY include/vantage DESTINATION include)
